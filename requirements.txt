# EaZi Platform — System Requirements & Dependencies
# ====================================================
# This file documents all system-level and project-level
# requirements needed to develop, build, and deploy the app.


# ── SYSTEM REQUIREMENTS ──────────────────────────────────────────────────────

# Node.js
# Version: 18.x LTS or higher (20.x LTS recommended)
# Install: https://nodejs.org
# Check:   node --version

# npm
# Version: 9.x or higher (bundled with Node.js)
# Check:   npm --version

# Git
# Version: 2.40 or higher
# Install: https://git-scm.com
# Check:   git --version

# Expo CLI (global, optional — npx usage preferred)
# Install: npm install -g expo-cli   OR use via npx expo
# Check:   npx expo --version

# Watchman (macOS/Linux only — improves file-watching performance)
# Install: brew install watchman
# Check:   watchman --version

# Android Studio (for Android emulator)
# Version: Hedgehog (2023.1.1) or higher
# Required: Android SDK, Android Virtual Device
# Install: https://developer.android.com/studio

# Xcode (macOS only — for iOS simulator)
# Version: 15 or higher
# Required: iOS Simulator, Command Line Tools
# Install: Mac App Store

# Supabase CLI (for local development and migrations)
# Version: 1.x or higher
# Install: npm install -g supabase   OR brew install supabase/tap/supabase
# Check:   supabase --version


# ── DEPLOYMENT REQUIREMENTS ──────────────────────────────────────────────────

# Vercel CLI (optional — Vercel dashboard is the primary deployment method)
# Install: npm install -g vercel
# Check:   vercel --version

# Vercel Account
# Sign up: https://vercel.com
# Project settings:
#   Framework Preset:  Other
#   Build Command:     npx expo export --platform web
#   Output Directory:  dist
#   Install Command:   npm install
#   Root Directory:    eazi-claude-code/eazi-app

# Supabase Account (for cloud database)
# Sign up: https://supabase.com
# Required env vars: EXPO_PUBLIC_SUPABASE_URL, EXPO_PUBLIC_SUPABASE_ANON_KEY


# ── ENVIRONMENT VARIABLES ─────────────────────────────────────────────────────
# Copy .env.example to .env and fill in the values.
# All vars prefixed with EXPO_PUBLIC_ are bundled into the client build.

# EXPO_PUBLIC_SUPABASE_URL       — Supabase project URL
# EXPO_PUBLIC_SUPABASE_ANON_KEY  — Supabase anonymous key (safe for client)
# EXPO_PUBLIC_GOOGLE_MAPS_API_KEY — Google Maps API key (Android/iOS only)
# EXPO_PUBLIC_YOCO_PUBLIC_KEY    — Yoco payments public key
# EXPO_PUBLIC_SENTRY_DSN         — Sentry error tracking DSN
